import Fastify from 'fastify';
import type { FastifyReply, FastifyRequest } from 'fastify';

interface Item {
  id: number;
  owner: string;
  name: string;
}

/** ユーザーと役割は「サーバー側」で管理する。クライアントが送る値を信用しない。 */
const USERS: Record<string, { role: 'user' | 'admin' }> = {
  alice: { role: 'user' },
  bob: { role: 'user' },
  root: { role: 'admin' },
};

export function buildLabApp() {
  const app = Fastify();
  const items: Item[] = [];

  // 認証: x-user が既知のユーザーであること(学習用の簡易認証)
  function authenticate(req: FastifyRequest, reply: FastifyReply) {
    const name = req.headers['x-user'];
    const user = typeof name === 'string' ? USERS[name] : undefined;
    if (typeof name !== 'string' || !user) {
      void reply.status(401).send({ error: 'unauthorized' });
      return null;
    }
    return { name, role: user.role };
  }

  // 認可: 持ち主か管理者のみ。他人のものは「存在しない」ことにする(存在の有無も隠す)
  function findAuthorized(id: string, user: { name: string; role: string }) {
    const item = items.find((i) => i.id === Number(id));
    return item && (item.owner === user.name || user.role === 'admin') ? item : undefined;
  }

  app.post('/items', async (req, reply) => {
    const user = authenticate(req, reply);
    if (!user) return;
    const { name } = req.body as { name: string };
    const item = { id: items.length + 1, owner: user.name, name };
    items.push(item);
    return reply.status(201).send(item);
  });

  app.get('/items/:id', async (req, reply) => {
    const user = authenticate(req, reply);
    if (!user) return;
    const item = findAuthorized((req.params as { id: string }).id, user);
    return item ?? reply.status(404).send({ error: 'not found' });
  });

  app.delete('/items/:id', async (req, reply) => {
    const user = authenticate(req, reply);
    if (!user) return;
    const item = findAuthorized((req.params as { id: string }).id, user);
    if (!item) return reply.status(404).send({ error: 'not found' });
    items.splice(items.indexOf(item), 1);
    return reply.status(204).send();
  });

  app.get('/admin/stats', async (req, reply) => {
    const user = authenticate(req, reply);
    if (!user) return;
    if (user.role !== 'admin') return reply.status(403).send({ error: 'forbidden' });
    return { count: items.length };
  });

  return app;
}
