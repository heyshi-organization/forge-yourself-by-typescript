import Fastify from 'fastify';
import { z } from 'zod';

const createSchema = z
  .object({
    name: z.string().regex(/^[A-Za-z0-9._-]{1,100}$/),
    visibility: z.enum(['public', 'private']),
  })
  .strict(); // 想定外のフィールド(status, owner など)は拒否する

const listSchema = z.object({ limit: z.coerce.number().int().min(1).max(100).default(10) }).strict();

interface StoredRequest {
  id: number;
  name: string;
  visibility: 'public' | 'private';
  status: 'PENDING';
  owner: string;
}

export function buildLabApp() {
  const app = Fastify({ bodyLimit: 16 * 1024 });
  const store: StoredRequest[] = [];

  app.post('/requests', async (req, reply) => {
    const parsed = createSchema.safeParse(req.body);
    if (!parsed.success) return reply.status(400).send({ error: 'invalid body' });
    // サーバー側で決める項目は、クライアントの入力から切り離して設定する
    const record: StoredRequest = { id: store.length + 1, status: 'PENDING', owner: 'anonymous', ...parsed.data };
    store.push(record);
    return reply.status(201).send(record);
  });

  app.get('/requests', async (req, reply) => {
    const parsed = listSchema.safeParse(req.query);
    if (!parsed.success) return reply.status(400).send({ error: 'invalid query' });
    return { items: store.slice(0, parsed.data.limit) };
  });

  return app;
}
