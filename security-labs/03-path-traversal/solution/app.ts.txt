import { readFileSync, realpathSync } from 'node:fs';
import { resolve, sep } from 'node:path';
import Fastify from 'fastify';

const TEMPLATE_NAME = /^[A-Za-z0-9][A-Za-z0-9._-]{0,99}$/;

export function buildLabApp(baseDir: string) {
  const app = Fastify();
  // 実体のパス(シンボリックリンクを解決したもの)で比較する
  const templatesDir = realpathSync(resolve(baseDir, 'templates'));

  app.get('/templates', async (req, reply) => {
    const { name } = req.query as { name?: unknown };
    // 対策1: ファイル名として使える文字だけを許可する("/" や ".." を含む名前は拒否)
    if (typeof name !== 'string' || !TEMPLATE_NAME.test(name)) {
      return reply.status(400).send({ error: 'invalid name' });
    }
    try {
      // 対策2(多層防御): シンボリックリンクを解決した後のパスが、templatesDir の内側にあることを確認する
      const target = realpathSync(resolve(templatesDir, name));
      if (!target.startsWith(templatesDir + sep)) {
        return reply.status(400).send({ error: 'invalid name' });
      }
      return { content: readFileSync(target, 'utf8') };
    } catch {
      return reply.status(404).send({ error: 'not found' });
    }
  });

  return app;
}
